A file read at compile time changes what the module computes, though no source module changes.
#!VERIFY
#!XFAIL a file read at compile time is not an input of the warm session, so the module stays green and keeps the old value (cache-reuse, After IC item 1)
#!FILE data.txt
abc
#!FILE main.ms
import { expansionFile, readFile } from "std/meta";
function beside(name: string): string {
	const self = expansionFile();
	return self.substring(0, self.lastIndexOf("/") + 1) + name;
}
const length = @comptime { return readFile(beside("data.txt")).length; };
console.log(`${length}`);
