In a chain main -> b -> a, an interface edit of a reaches b; b's own interface is unchanged, so
main is not rechecked.
#!FILE a.ms
export function base(): int32 {
	return 1;
}
#!FILE b.ms
import { base } from "./a";
export function mid(): string {
	return `${base() + 10}`;
}
#!FILE main.ms
import { mid } from "./b";
console.log(mid());
#!STEP expect: 11
#!FILE a.ms
export function base(): int32 {
	return 7;
}
#!STEP expect: 17; body-edit
#!FILE a.ms
export function base(): int64 {
	return 7;
}
#!STEP expect: 17; iface-edit; rechecked: a.ms b.ms
