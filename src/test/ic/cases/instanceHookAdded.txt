A hook bound to one instance of a generic class, added and removed in the class's module, changes how
an importer holding that instance releases it, though the importer and the exports do not change.
#!FILE a.ms
let trail = "";
export function trailOf(): string {
	return trail;
}
export class Bag<T> {
	n: T;
	constructor(n: T) { this.n = n; }
}
#!FILE main.ms
import { Bag, trailOf } from "./a";
function run(): void {
	const b = new Bag<int32>(3);
}
run();
console.log(`end${trailOf()}`);
#!STEP expect: end

	constructor(n: T) { this.n = n; }

function onDestroy(this self: Bag<int32>): void {
	trail = trail + " d";
}
