main's call although b's interface text is unchanged.

#!FILE a.ms
export const N: int32 = 4;
#!FILE b.ms

	return x + 1;
}
#!FILE main.ms
import { f } from "./b";
console.log(f());

#!FILE a.ms
export const N: int32 = 6;
