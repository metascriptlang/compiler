import { hash } from "std/hash";
import { blake2bSync } from "std/crypto/blake2b";
import { ed25519FromSeed } from "std/crypto/ed25519";

export function label(): string {
	const key = ed25519FromSeed("a".repeat(32));
	const digest = blake2bSync("native first use");
	return ((digest.ok ? digest.value.length : 0) + (hash("abc") > 0 ? 0 : 1)).toString() + (key.ok ? " native pair" : " native failed");
}
