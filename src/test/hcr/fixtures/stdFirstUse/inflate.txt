import { hash } from "std/hash";
import { blake2bSync } from "std/crypto/blake2b";
import { ed25519FromSeed } from "std/crypto/ed25519";
import { gzip } from "std/compress/deflate";
import { gunzip } from "std/compress/inflate";

export function label(): string {
	const key = ed25519FromSeed("a".repeat(32));
	const packed = gzip("native first use");
	const nativeReady = hash("abc") > 0 && blake2bSync("native first use").ok && key.ok;
	if (!nativeReady || !packed.ok) return "native failed";
	const unpacked = gunzip(packed.value);
	return unpacked.ok && unpacked.value.asString() == "native first use" ? "inflate roundtrip" : "native failed";
}
