import { hash } from "std/hash";
import { blake2bSync } from "std/crypto/blake2b";
import { Ed25519PrivateKey } from "std/crypto/ed25519";
import { gzip } from "std/compress/deflate";
import { gunzip } from "std/compress/inflate";
import { jsonArray, jsonString, jsonNumber } from "std/serialize/json/builder";
import { JsonKind } from "std/serialize/json/types";

const key = await Ed25519PrivateKey.fromSeed("a".repeat(32));

export function label(): string {
	const packed = gzip("native first use");
	if (hash("abc") <= 0 || !blake2bSync("native first use").ok || !key.ok || !packed.ok) return "native failed";
	const unpacked = gunzip(packed.value);
	if (!unpacked.ok || unpacked.value.asString() != "native first use") return "native failed";
	const value = jsonArray([jsonArray([jsonString("json tag "), jsonNumber(7)])]);
	if (value.kind != JsonKind.Array) return "union failed";
	const nested = value.items[0];
	if (nested.kind != JsonKind.Array) return "union failed";
	const text = nested.items[0];
	const number = nested.items[1];
	if (text.kind != JsonKind.String || number.kind != JsonKind.Number) return "union failed";
	return text.strVal + number.numVal.toString();
}
